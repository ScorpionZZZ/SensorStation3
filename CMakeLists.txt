# The following five lines of boilerplate have to be in your project's
# CMakeLists in this exact order for cmake to work correctly
cmake_minimum_required(VERSION 3.22)

# ── App version ────────────────────────────────────────────────────────────
# All three fields are manual. Bump APP_VERSION_BUILD (and MAJOR/MINOR when
# appropriate) by hand for each release build.
set(APP_VERSION_MAJOR 1)
set(APP_VERSION_MINOR 6)
set(APP_VERSION_BUILD 8)

set(PROJECT_VER "${APP_VERSION_MAJOR}.${APP_VERSION_MINOR}.${APP_VERSION_BUILD}")
# ── End version ────────────────────────────────────────────────────────────

# Extra component directories searched by ESP-IDF build system.
# All subdirectories of components/ that contain a CMakeLists.txt are auto-discovered.
set(EXTRA_COMPONENT_DIRS "${CMAKE_CURRENT_SOURCE_DIR}/components")

# Must be set before components are processed so the LVGL managed component
# finds our config file instead of trying to read Kconfig.
set(LV_CONF_PATH
    "${CMAKE_CURRENT_SOURCE_DIR}/components/display/include/lv_conf.h"
    CACHE STRING "" FORCE)
set(LV_KCONFIG_IGNORE 1 CACHE STRING "" FORCE)

include($ENV{IDF_PATH}/tools/cmake/project.cmake)
project(SensorStation3)

# Generate build/config/build_info.h and build/${PROJECT_NAME}.json at configure
# time from the manual version numbers above. This reruns whenever CMakeLists.txt
# changes (i.e. when a version number is actually bumped), not on every build.
file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/config")
file(WRITE "${CMAKE_BINARY_DIR}/config/build_info.h"
    "#pragma once\n"
    "/* Auto-generated by CMakeLists.txt from APP_VERSION_* — do not edit */\n"
    "#define APP_VERSION_MAJOR ${APP_VERSION_MAJOR}\n"
    "#define APP_VERSION_MINOR ${APP_VERSION_MINOR}\n"
    "#define APP_BUILD_NUMBER  ${APP_VERSION_BUILD}\n"
    "#define APP_VERSION_STR   \"${PROJECT_VER}\"\n"
)
file(WRITE "${CMAKE_BINARY_DIR}/${CMAKE_PROJECT_NAME}.json"
    "{\"version\":\"${PROJECT_VER}\"}\n"
)
